Frequently Asked Questions

Everything you need to know about NutriSmart AI and capstone specs.

How does the AI image recognition work?

NutriSmart utilizes a Python FastAPI backend integrated with a custom computer vision model. For production scalability, the API architecture exposes clean decoupling interfaces to plug in TensorFlow or OpenAI Vision APIs seamlessly.

Are prices accurate for my campus?

Prices are averaged across standard supermarket items and local student cafeteria meal options. You can adjust your maximum budget threshold using the interactive slider.

What backend database schema is used?

The system connects to MySQL via SQLAlchemy ORM. Tables include `users`, `foods`, `nutrition`, `recommendations`, and `food_preferences` with relational constraints. Click "Source Code & Architecture" to review full SQL files.

from fastapi import FastAPI, UploadFile, File, Form, Depends, HTTPException, status
from fastapi.middleware.cors import CORSMiddleware

from typing import List, Optional

from database.db import get_db, engine, Base

app = FastAPI(
    title="NutriSmart API",
    description="Smart Nutrition & Meal Recommendation System for University Students",
    version="1.0.0"
)

# CORS setup for web frontend communication
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

recommender = RecommendationEngine()

@app.get("/")
def read_root():
    return {"system": "NutriSmart API", "status": "Online", "version": "1.0.0"}

@app.post("/api/analyze-food", response_model=MealResponse)
async def analyze_food(
    file: Optional[UploadFile] = File(None),
    budget: float = Form(7.50),
    time_limit: int = Form(15),
    preference: str = Form("balanced"),
    db: Session = Depends(get_db)
):
    """
    Analyzes uploaded image or preset and returns nutrition metrics & recommendations.
    """
    if file:
        # File type validation
        allowed_types = ["image/jpeg", "image/png", "image/webp"]
        if file.content_type not in allowed_types:
            raise HTTPException(status_code=400, detail="Invalid file type. Only JPEG, PNG, WEBP allowed.")

    # Call recommendation modular service
    result = recommender.analyze_and_recommend(
        budget=budget, 
        max_time=time_limit, 
        preference=preference
    )
    return result

@app.get("/api/foods", response_model=List[FoodItem])
def get_all_foods(db: Session = Depends(get_db)):
    return recommender.get_all_student_meals()

if __name__ == "__main__":
    uvicorn.run("main:app", host="0.0.0.0", port=8000, reload=True)

from sqlalchemy import Column, Integer, String, Float, ForeignKey, Text, DateTime
from sqlalchemy.orm import relationship, declarative_base
from datetime import datetime

Base = declarative_base()

class User(Base):
    __tablename__ = "users"
    id = Column(Integer, primary_key=True, index=True)
    student_id = Column(String(50), unique=True, index=True, nullable=False)
    name = Column(String(100), nullable=False)
    email = Column(String(100), unique=True, nullable=False)
    daily_budget = Column(Float, default=10.0)

class Food(Base):
    __tablename__ = "foods"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(150), nullable=False)
    category = Column(String(50))
    image_url = Column(String(255))
    price = Column(Float, nullable=False)
    prep_time = Column(Integer, nullable=False) # In minutes
    
    nutrition = relationship("Nutrition", back_populates="food", uselist=False)

class Nutrition(Base):
    __tablename__ = "nutrition"
    id = Column(Integer, primary_key=True, index=True)
    food_id = Column(Integer, ForeignKey("foods.id"), nullable=False)
    calories = Column(Float, nullable=False)
    protein = Column(Float, nullable=False)
    carbohydrates = Column(Float, nullable=False)
    fats = Column(Float, nullable=False)

    food = relationship("Food", back_populates="nutrition")

# Rule-Based Recommendation Engine with ML Interface hook
class RecommendationEngine:
    def __init__(self):
        # Initial dataset simulating MySQL DB query
        self.catalog = [
            {
                "id": 1, "name": "Mediterranean Quinoa Bowl", "price": 5.50, "prep_time": 10,
                "category": "high-protein", "image_url": "https://images.unsplash.com/photo-1546069901-ba9599a7e63c",
                "nutrition": {"calories": 520, "protein": 24, "carbohydrates": 62}
            },
            {
                "id": 2, "name": "Grilled Chicken & Rice Box", "price": 6.80, "prep_time": 12,
                "category": "high-protein", "image_url": "https://images.unsplash.com/photo-1532550907401-a500c9a57435",
                "nutrition": {"calories": 580, "protein": 38, "carbohydrates": 54}
            },
            {
                "id": 3, "name": "Peanut Butter & Banana Oats", "price": 2.80, "prep_time": 5,
                "category": "breakfast", "image_url": "https://images.unsplash.com/photo-1517673400267-0251440c45dc",
                "nutrition": {"calories": 410, "protein": 14, "carbohydrates": 58}
            }
        ]

    def analyze_and_recommend(self, budget: float, max_time: int, preference: str):
        # Modular scoring algorithm
        filtered = []
        for item in self.catalog:
            if item["price"] <= budget and item["prep_time"] <= max_time:
                match_score = 100.0 - ((item["price"] / budget) * 10)
                reason = f"Fits budget (₩{item['price']}) & prep under {max_time} mins."
                filtered.append({
                    "food": item,
                    "match_score": round(match_score, 1),
                    "reason": reason
                })
        return {
            "analyzed_meal": self.catalog[0],
            "recommendations": filtered
        }
File: database/schema.sql (Production MySQL Script) 
CREATE DATABASE IF NOT EXISTS nutrismart_db;
USE nutrismart_db;

-- Table: Users
CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    student_id VARCHAR(50) NOT NULL UNIQUE,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Table: Foods
CREATE TABLE IF NOT EXISTS foods (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    category VARCHAR(50) NOT NULL,
    price DECIMAL(6,2) NOT NULL,
    prep_time INT NOT NULL,
    image_url VARCHAR(255)
);

-- Table: Nutrition
CREATE TABLE IF NOT EXISTS nutrition (
    id INT AUTO_INCREMENT PRIMARY KEY,
    food_id INT UNIQUE,
    calories FLOAT NOT NULL,
    protein FLOAT NOT NULL,
    carbohydrates FLOAT NOT NULL,
    fats FLOAT NOT NULL,
    FOREIGN KEY (food_id) REFERENCES foods(id) ON DELETE CASCADE
);
